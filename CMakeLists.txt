cmake_minimum_required(VERSION 3.24)
project(ThinkerCad VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(THINKERCAD_PUBLISHER_SHA256 "" CACHE STRING "SHA256 fingerprint of the trusted Windows signing certificate")
option(THINKERCAD_GUI "Build the Qt desktop application" ON)
include(CTest)
add_library(thinkercad_core src/core/simulation.cpp src/core/circuit.cpp)
target_include_directories(thinkercad_core PUBLIC include)
if(MSVC)
  target_compile_options(thinkercad_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(thinkercad_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(thinkercad_demo examples/core_demo.cpp)
target_link_libraries(thinkercad_demo PRIVATE thinkercad_core)
add_executable(sketch_example sdk/runner.cpp)
target_include_directories(sketch_example PRIVATE sdk)
target_compile_definitions(sketch_example PRIVATE THINKERCAD_SIMULATION=1 THINKERCAD_SKETCH="${CMAKE_CURRENT_SOURCE_DIR}/examples/animation.ino")
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE thinkercad_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(circuit_tests tests/circuit_tests.cpp)
  target_link_libraries(circuit_tests PRIVATE thinkercad_core)
  add_test(NAME circuit COMMAND circuit_tests)
  add_executable(sdk_flash_tests tests/sdk_flash_tests.cpp)
  target_include_directories(sdk_flash_tests PRIVATE sdk)
  add_test(NAME sdk_flash COMMAND sdk_flash_tests)
  add_test(NAME arduino_animation COMMAND sketch_example --iterations 2)
  set_tests_properties(arduino_animation PROPERTIES PASS_REGULAR_EXPRESSION "FRAME [0-9a-f]+" TIMEOUT 5)
endif()
if(THINKERCAD_GUI)
  set(CMAKE_INSTALL_BINDIR ".") # Flat desktop install: executable, SDK and runtime DLLs share a root.
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network)
  qt_standard_project_setup()
  add_library(thinkercad_desktop STATIC src/gui/mainwindow.cpp src/gui/mainwindow.hpp src/gui/codeinterpreter.cpp src/gui/codeinterpreter.hpp src/gui/updater.cpp src/gui/updater.hpp src/gui/circuitcanvas.cpp src/gui/circuitcanvas.hpp src/gui/codeeditor.cpp src/gui/codeeditor.hpp src/gui/projectio.cpp src/gui/projectio.hpp src/gui/examples.cpp src/gui/examples.hpp)
  target_include_directories(thinkercad_desktop PUBLIC src/gui)
  target_link_libraries(thinkercad_desktop PUBLIC thinkercad_core Qt6::Widgets Qt6::Network)
  target_compile_definitions(thinkercad_desktop PUBLIC THINKERCAD_VERSION="${PROJECT_VERSION}" THINKERCAD_PUBLISHER_SHA256="${THINKERCAD_PUBLISHER_SHA256}")
  qt_add_executable(ThinkerCad WIN32 src/gui/main.cpp)
  target_link_libraries(ThinkerCad PRIVATE thinkercad_desktop)
  qt_add_resources(ThinkerCad branding PREFIX "/brand" BASE assets FILES assets/logo.png)
  target_compile_definitions(ThinkerCad PRIVATE THINKERCAD_VERSION="${PROJECT_VERSION}" THINKERCAD_PUBLISHER_SHA256="${THINKERCAD_PUBLISHER_SHA256}")
  if(WIN32)
    target_link_libraries(thinkercad_desktop PUBLIC wintrust crypt32)
    enable_language(RC)
    configure_file(packaging/windows/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
    target_sources(ThinkerCad PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
    add_executable(ThinkerCadUpdater WIN32 packaging/windows/update_helper.cpp)
    target_compile_definitions(ThinkerCadUpdater PRIVATE THINKERCAD_PUBLISHER_SHA256="${THINKERCAD_PUBLISHER_SHA256}")
    target_link_libraries(ThinkerCadUpdater PRIVATE wintrust crypt32 shell32)
    if(MSVC)
      set_property(TARGET ThinkerCadUpdater PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    elseif(MINGW)
      target_link_options(ThinkerCadUpdater PRIVATE -municode -static)
    endif()
    install(TARGETS ThinkerCadUpdater RUNTIME DESTINATION .)
  endif()
  file(COPY sdk examples DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
  if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(desktop_tests tests/desktop_tests.cpp)
    target_link_libraries(desktop_tests PRIVATE thinkercad_desktop Qt6::Test)
    qt_add_resources(desktop_tests test_branding PREFIX "/brand" BASE assets FILES assets/logo.png)
    add_test(NAME desktop_interactions COMMAND desktop_tests)
    set_tests_properties(desktop_interactions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
    if(WIN32)
      add_test(NAME windows_native_chrome COMMAND desktop_tests nativeWindowChrome)
      set_tests_properties(windows_native_chrome PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows" TIMEOUT 30)
    endif()
    qt_add_executable(interpreter_tests tests/interpreter_tests.cpp src/gui/codeinterpreter.cpp src/gui/codeinterpreter.hpp)
    target_include_directories(interpreter_tests PRIVATE src/gui)
    target_link_libraries(interpreter_tests PRIVATE Qt6::Core)
    add_test(NAME library_and_ipc COMMAND interpreter_tests)
    set_tests_properties(library_and_ipc PROPERTIES TIMEOUT 30)
    qt_add_executable(speed_tests tests/speed_tests.cpp)
    target_link_libraries(speed_tests PRIVATE thinkercad_desktop)
    add_test(NAME live_simulation_speed COMMAND speed_tests)
    set_tests_properties(live_simulation_speed PROPERTIES TIMEOUT 30)
    add_test(NAME desktop_smoke COMMAND ThinkerCad --smoke-test)
    set_tests_properties(desktop_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  endif()
  install(TARGETS ThinkerCad RUNTIME DESTINATION .)
  install(DIRECTORY sdk DESTINATION .)
  install(DIRECTORY examples/ DESTINATION examples FILES_MATCHING PATTERN "*.ino")
  qt_generate_deploy_app_script(TARGET ThinkerCad OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
  # Debian's generic runtime deployment does not copy QPA plugins automatically.
  if(UNIX AND NOT APPLE)
    foreach(plugin QOffscreenIntegrationPlugin QXcbIntegrationPlugin)
      find_package(Qt6${plugin} REQUIRED PATHS "${Qt6Gui_DIR}" NO_DEFAULT_PATH)
      if(TARGET Qt6::${plugin})
        install(FILES $<TARGET_FILE:Qt6::${plugin}> DESTINATION plugins/platforms)
      endif()
    endforeach()
  endif()
endif()
